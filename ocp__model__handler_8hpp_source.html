40#define ACADOS_DISPATCH(function_name, ...) \
41 if (std::holds_alternative<karl_solver_capsule*>(capsule)) { \
42 return karl_##function_name(std::get<karl_solver_capsule*>(capsule), ##__VA_ARGS__); \
43 } else if (std::holds_alternative<shuttle_solver_capsule*>(capsule)) { \
44 return shuttle_##function_name(std::get<shuttle_solver_capsule*>(capsule), ##__VA_ARGS__); \
45 } else { \
46 throw std::invalid_argument("Invalid capsule type."); \
47 }
