17TrajectoryOptimizationNode::TrajectoryOptimizationNode(const std::string node_name, const rclcpp::NodeOptions& options)
18 : rclcpp::Node(node_name, options) {
19 // declare and load node parameters
20 this->declareAndLoadParameter("vehicle_frame_id", vehicle_frame_id_,
21 "Frame ID of local vehicle frame (the ocp is defined in this frame)");
22 this->declareAndLoadParameter("trajectory_frame_id", trajectory_frame_id_, "Frame ID of output trajectory");
23 this->declareAndLoadParameter("fixed_over_time_frame_id", fixed_over_time_frame_id_,
24 "Frame ID of frame that is fixed over time for finding temporal transforms");
25 this->declareAndLoadParameter("ego_data_timeout", ego_data_timeout_,
26 "Time after which a received ego vehicle data is considered invalid [s]. Optimization will not "
27 "be run if ego data is invalid.");
28 this->declareAndLoadParameter("model_name", model_name_,
29 "Name of the model to be used for trajectory optimization [karl, shuttle]");
30 this->declareAndLoadParameter("optimization_frequency", optimization_freq_, "Optimization frequency in Hz");
31 this->declareAndLoadParameter("n_shots", n_shots_, "Number of shooting intervals in optimization horizon");
32 this->declareAndLoadParameter("optimization_horizon", optimization_horizon_, "Optimization Horizon in seconds");
33 this->declareAndLoadParameter("verbose", verbose_, "Print solver statistics");
34 this->declareAndLoadParameter("debug_visualization", debug_viz_, "Publish debug visualization markers (e.g. obstacle circles)");
35 this->declareAndLoadParameter("run_as_callback", run_as_callback_,
36 "Run OCP once for each received reference trajectory (true) or on a timer (false)");
37 this->declareAndLoadParameter("cost_weights", cost_weights_, "Cost function weights");
38 this->declareAndLoadParameter("dynamic_weight", dynamic_weight_, "Dynamic weight alpha");
39 this->declareAndLoadParameter("thw", thw_, "Time headway to front vehicle");
40 this->declareAndLoadParameter("d_min_obstacle_long", d_min_obstacle_long_,
41 "Minimum distance to keep to obstacle in longitudinal direction [m]");
42 this->declareAndLoadParameter("d_min_obstacle_lat", d_min_obstacle_lat_,
43 "Minimum distance to keep to obstacle in lateral direction [m]");
44 this->declareAndLoadParameter("d_min_boundary_lat", d_min_boundary_lat_,
45 "Minimum distance to keep to boundary in lateral direction [m]");
46 this->declareAndLoadParameter("standstill_threshold", standstill_threshold_,
47 "Threshold for standstill detection [m/s]. If the velocities of all states are below this "
48 "threshold, publish standstill trajectory");
49 this->declareAndLoadParameter("high_level_stabilization", high_level_stabilization_,
50 "Use high-level stabilization strategy for init state (= init with current EgoData)");
51 this->declareAndLoadParameter(
52 "add_x_init_to_ref", add_x_init_to_ref_,
53 "add initial state of OCP to beginning of reference trajectory if this starts in front of ego vehicle");
54 this->declareAndLoadParameter(
55 "consider_objects", consider_objects_,
56 "consider objects in optimization: 0 = none, 1 = static (no prediction), 2 = dynamic (with prediction)");
57 this->declareAndLoadParameter(
58 "consider_boundaries", consider_boundaries_,
59 "consider route boundaries in optimization: 0 = no, 1 = suggested lane, 2 = including adjacent, 3 = drivable space");
60 this->declareAndLoadParameter("bi_level_dV", bi_level_dV_,
61 "Threshold for bi-level stabilization: maximum velocity difference [m/s]");
62 this->declareAndLoadParameter("bi_level_dA", bi_level_dA_,
63 "Threshold for bi-level stabilization: maximum acceleration difference [m/s^2]");
64 this->declareAndLoadParameter("bi_level_dY", bi_level_dY_, "Threshold for bi-level stabilization: maximum y-offset [m]");
65 this->declareAndLoadParameter("bi_level_dYaw", bi_level_dYaw_,
66 "Threshold for bi-level stabilization: maximum yaw difference [degree]");
67 this->declareAndLoadParameter(
68 "init_as_ref", init_as_ref_,
69 "Boolean that enables initialization of trajectory states as reference states under certain set of conditions");
70 this->setup();
71}

75template <typename T>
76void TrajectoryOptimizationNode::declareAndLoadParameter(const std::string& name,
77 T& param,
78 const std::string& description,
79 const bool add_to_auto_reconfigurable_params,
80 const bool is_required,
81 const bool read_only,
82 const std::optional<double>& from_value,
83 const std::optional<double>& to_value,
84 const std::optional<double>& step_value,
85 const std::string& additional_constraints) {
86 rcl_interfaces::msg::ParameterDescriptor param_desc;
87 param_desc.description = description;
88 param_desc.additional_constraints = additional_constraints;
89 param_desc.read_only = read_only;
90
91 auto type = rclcpp::ParameterValue(param).get_type();
92
93 if (from_value.has_value() && to_value.has_value()) {
94 if constexpr (std::is_integral_v<T>) {
95 rcl_interfaces::msg::IntegerRange range;
96 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value()));
97 if (step_value.has_value()) range.set__step(static_cast<T>(step_value.value()));
98 param_desc.integer_range = {range};
99 } else if constexpr (std::is_floating_point_v<T>) {
100 rcl_interfaces::msg::FloatingPointRange range;
101 range.set__from_value(static_cast<T>(from_value.value())).set__to_value(static_cast<T>(to_value.value()));
102 if (step_value.has_value()) range.set__step(static_cast<T>(step_value.value()));
103 param_desc.floating_point_range = {range};
104 } else {
105 RCLCPP_WARN(this->get_logger(), "Parameter type of parameter '%s' does not support specifying a range", name.c_str());
106 }
107 }
108
109 this->declare_parameter(name, type, param_desc);
110
111 try {
112 param = this->get_parameter(name).get_value<T>();
113 std::stringstream ss;
114 ss << "Loaded parameter '" << name << "': ";
115 if constexpr (is_vector_v<T>) {
116 ss << "[";
117 for (const auto& element : param) ss << element << (&element != &param.back() ? ", " : "");
118 ss << "]";
119 } else {
120 ss << param;
121 }
122 RCLCPP_INFO_STREAM(this->get_logger(), ss.str());
123 } catch (rclcpp::exceptions::ParameterUninitializedException&) {
124 if (is_required) {
125 RCLCPP_FATAL_STREAM(this->get_logger(), "Missing required parameter '" << name << "', exiting");
126 exit(EXIT_FAILURE);
127 } else {
128 std::stringstream ss;
129 ss << "Missing parameter '" << name << "', using default value: ";
130 if constexpr (is_vector_v<T>) {
131 ss << "[";
132 for (const auto& element : param) ss << element << (&element != &param.back() ? ", " : "");
133 ss << "]";
134 } else {
135 ss << param;
136 }
137 RCLCPP_WARN_STREAM(this->get_logger(), ss.str());
138 this->set_parameters({rclcpp::Parameter(name, rclcpp::ParameterValue(param))});
139 }
140 }
141
142 if (add_to_auto_reconfigurable_params) {
143 std::function<void(const rclcpp::Parameter&)> setter = [&param](const rclcpp::Parameter& p) { param = p.get_value<T>(); };
144 auto_reconfigurable_params_.push_back(std::make_tuple(name, setter));
145 }
146}

148rcl_interfaces::msg::SetParametersResult TrajectoryOptimizationNode::parametersCallback(
149 const std::vector<rclcpp::Parameter>& parameters) {
150 for (const auto& param : parameters) {
151 for (auto& auto_reconfigurable_param : auto_reconfigurable_params_) {
152 if (param.get_name() == std::get<0>(auto_reconfigurable_param)) {
153 std::get<1>(auto_reconfigurable_param)(param);
154 RCLCPP_INFO(this->get_logger(), "Reconfigured parameter '%s' to: %s", param.get_name().c_str(),
155 param.value_to_string().c_str());
156 break;
157 }
158 }
159 // handle special cases
160 if (param.get_name() == "run_as_callback") {
161 if (!run_as_callback_ && !planning_timer_) {
162 planning_timer_ = this->create_wall_timer(std::chrono::duration<double>(1 / optimization_freq_),
163 std::bind(&TrajectoryOptimizationNode::planningCycle, this));
164 RCLCPP_WARN(this->get_logger(), "OCP runs now periodically with frequency %f Hz", optimization_freq_);
165 } else if (run_as_callback_ && planning_timer_) {
166 planning_timer_->cancel();
167 planning_timer_.reset();
168 RCLCPP_WARN(this->get_logger(), "OCP runs now on reference trajectory callback");
169 }
170 }
171 }
172 // mark parameter change successful
173 rcl_interfaces::msg::SetParametersResult result;
174 result.successful = true;
175
176 return result;
177}

179void TrajectoryOptimizationNode::setup() {
180 tf2_buffer_ = std::make_unique<tf2_ros::Buffer>(this->get_clock());
181 tf2_listener_ = std::make_shared<tf2_ros::TransformListener>(*tf2_buffer_);
182
183 // create a callback for dynamic parameter configuration
184 parameters_callback_ = this->add_on_set_parameters_callback(
185 std::bind(&TrajectoryOptimizationNode::parametersCallback, this, std::placeholders::_1));
186
187 // set up subscriber for input topics
188 ego_data_sub_ = this->create_subscription<perception_msgs::msg::EgoData>(
189 "~/ego_data", 1, std::bind(&TrajectoryOptimizationNode::egoDataCallback, this, std::placeholders::_1));
190 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", ego_data_sub_->get_topic_name());
191
192 object_list_sub_ = this->create_subscription<perception_msgs::msg::ObjectList>(
193 "~/object_list", 1, std::bind(&TrajectoryOptimizationNode::objectListCallback, this, std::placeholders::_1));
194 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", object_list_sub_->get_topic_name());
195
196 route_sub_ = this->create_subscription<route_planning_msgs::msg::Route>(
197 "~/route", 1, std::bind(&TrajectoryOptimizationNode::routeCallback, this, std::placeholders::_1));
198 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", route_sub_->get_topic_name());
199
200 reference_trajectory_sub_ = this->create_subscription<trajectory_planning_msgs::msg::Trajectory>(
201 "~/reference_trajectory", 1,
202 std::bind(&TrajectoryOptimizationNode::referenceTrajectoryCallback, this, std::placeholders::_1));
203 RCLCPP_INFO(this->get_logger(), "Subscribed to '%s'", reference_trajectory_sub_->get_topic_name());
204
205 // set up publisher for output topics
206 trajectory_pub_ = this->create_publisher<trajectory_planning_msgs::msg::Trajectory>("~/trajectory", 1);
207 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", trajectory_pub_->get_topic_name());
208 circles_pub_ = this->create_publisher<visualization_msgs::msg::MarkerArray>("~/visualization/object_circles", 1);
209 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", circles_pub_->get_topic_name());
210 ego_circles_pub_ = this->create_publisher<visualization_msgs::msg::MarkerArray>("~/visualization/ego_circles", 1);
211 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", ego_circles_pub_->get_topic_name());
212 boundary_pub_ = this->create_publisher<visualization_msgs::msg::MarkerArray>("~/visualization/boundaries", 1);
213 RCLCPP_INFO(this->get_logger(), "Publishing to '%s'", boundary_pub_->get_topic_name());
214
215 // create timer for planning cycle
216 if (run_as_callback_) {
217 RCLCPP_INFO(this->get_logger(), "OCP runs on reference trajectory callback");
218 } else {
219 RCLCPP_INFO(this->get_logger(), "OCP runs continuously with frequency %f Hz", optimization_freq_);
220 planning_timer_ = this->create_wall_timer(std::chrono::duration<double>(1 / optimization_freq_),
221 std::bind(&TrajectoryOptimizationNode::planningCycle, this));
222 }
223
224 // init reference trajectory
225 trajectory_planning_msgs::trajectory_access::initializeTrajectory(
226 reference_trajectory_, trajectory_planning_msgs::msg::REFERENCE::TYPE_ID, n_shots_ + 1);
227 reference_trajectory_.header.frame_id = vehicle_frame_id_;
228
229 // init latest trajectory (doesn't matter which type, only used for standstill detection)
230 trajectory_planning_msgs::trajectory_access::initializeTrajectory(
231 latest_valid_trajectory_, trajectory_planning_msgs::msg::DRIVABLE::TYPE_ID, n_shots_ + 1);
232 latest_valid_trajectory_.header.frame_id = trajectory_frame_id_;
233
234 setupSolver();
235
236 // Annotate message links for tracing: Publish trajectory periodically, which depends an all subscribed topics.
237 std::vector<const void*> link_subs;
238 link_subs.push_back(static_cast<const void*>(ego_data_sub_->get_subscription_handle().get()));
239 link_subs.push_back(static_cast<const void*>(object_list_sub_->get_subscription_handle().get()));
240 link_subs.push_back(static_cast<const void*>(route_sub_->get_subscription_handle().get()));
241 link_subs.push_back(static_cast<const void*>(reference_trajectory_sub_->get_subscription_handle().get()));
242 std::vector<const void*> link_pubs;
243 link_pubs.push_back(static_cast<const void*>(trajectory_pub_->get_publisher_handle().get()));
244 RCLCPP_INFO(get_logger(), "Annotating message links for tracing with %zu subscriptions and %zu publications", link_subs.size(),
245 link_pubs.size());
246 TRACETOOLS_TRACEPOINT(message_link_periodic_async, link_subs.data(), link_subs.size(), link_pubs.data(), link_pubs.size());
247}

249void TrajectoryOptimizationNode::setupSolver() {
250 // setup acados solver
251 ocp_capsule_ = trajectory_optimization::acados_create_capsule(model_name_);
252 int status = trajectory_optimization::acados_create(ocp_capsule_);
253 nlp_dims_ = trajectory_optimization::acados_get_nlp_dims(ocp_capsule_);
254 if (n_shots_ <= 0) {
255 RCLCPP_FATAL(this->get_logger(), "n_shots must be > 0, got %d", n_shots_);
256 exit(1);
257 } else if (n_shots_ != nlp_dims_->N) {
258 std::vector<double> new_time_steps(n_shots_, optimization_horizon_ / n_shots_);
259 RCLCPP_INFO(this->get_logger(), "Recreate OCP with: horizon = %f, n_shots = %d, dt = %f", optimization_horizon_, n_shots_,
260 new_time_steps.front());
261 status = trajectory_optimization::acados_create_with_discretization(ocp_capsule_, n_shots_, new_time_steps.data());
262 }
263
264 if (status) {
265 RCLCPP_INFO(this->get_logger(), "%s_acados_create_with_discretization() returned status %d. Exiting.", model_name_.c_str(),
266 status);
267 exit(1);
268 }
269
270 nlp_config_ = trajectory_optimization::acados_get_nlp_config(ocp_capsule_);
271 nlp_dims_ = trajectory_optimization::acados_get_nlp_dims(ocp_capsule_);
272 nlp_in_ = trajectory_optimization::acados_get_nlp_in(ocp_capsule_);
273 nlp_out_ = trajectory_optimization::acados_get_nlp_out(ocp_capsule_);
274 nlp_solver_ = trajectory_optimization::acados_get_nlp_solver(ocp_capsule_);
275 nlp_opts_ = trajectory_optimization::acados_get_nlp_opts(ocp_capsule_);
276
277 // initialization of state and control values; set all to zero
278 std::vector<double> x_init(*nlp_dims_->nx, 0.0);
279 std::vector<double> u_init(*nlp_dims_->nu, 0.0);
280
281 // initialize solution
282 for (int i = 0; i < n_shots_; ++i) {
283 ocp_nlp_out_set(nlp_config_, nlp_dims_, nlp_out_, nlp_in_, i, "x", x_init.data());
284 ocp_nlp_out_set(nlp_config_, nlp_dims_, nlp_out_, nlp_in_, i, "u", u_init.data());
285 }
286 ocp_nlp_out_set(nlp_config_, nlp_dims_, nlp_out_, nlp_in_, n_shots_, "x", x_init.data());
287
288 xtraj_.resize(*nlp_dims_->nx * (n_shots_ + 1));
289 utraj_.resize(*nlp_dims_->nu * n_shots_);
290}

292void TrajectoryOptimizationNode::freeSolver() {
293 // free solver
294 int status = trajectory_optimization::acados_free(ocp_capsule_);
295 if (status) {
296 RCLCPP_ERROR(this->get_logger(), "%s_acados_free() returned status %d.", model_name_.c_str(), status);
297 }
298 // free solver capsule
299 status = trajectory_optimization::acados_free_capsule(ocp_capsule_);
300 if (status) {
301 RCLCPP_ERROR(this->get_logger(), "%s_acados_free_capsule() returned status %d.", model_name_.c_str(), status);
302 }
303}

310void TrajectoryOptimizationNode::planningCycle() {
311 if (debug_viz_) viz_circles_.clear();
312 if (rclcpp::Time(this->now()) - rclcpp::Time(ego_data_.header.stamp) > rclcpp::Duration::from_seconds(ego_data_timeout_)) {
313 RCLCPP_WARN(this->get_logger(), "EgoData outdated. Skipping planning cycle.");
314 return;
315 }
316 // init trajectory message and set header
317 trajectory_planning_msgs::msg::Trajectory::UniquePtr trajectory = std::make_unique<trajectory_planning_msgs::msg::Trajectory>();
318 initializeTrajectory(*trajectory);
319
320 trajectory->header.frame_id = vehicle_frame_id_;
321 trajectory->header.stamp = ego_data_.header.stamp; // use latest ego_data stamp as trajectory stamp
322
323 // init time-steps of trajectory to ensure increasing time-steps even for standstill trajectories
324 double dt = optimization_horizon_ / n_shots_;
325 for (int i = 0; i <= n_shots_; ++i) trajectory_planning_msgs::trajectory_access::setT(*trajectory, i * dt, i);
326
327 // check if the reference trajectory is standstill
328 if (trajectory_planning_msgs::trajectory_access::getStandstill(reference_trajectory_)) {
329 RCLCPP_WARN(this->get_logger(), "Standstill trajectory. Skipping planning cycle. Publish standstill trajectory.");
330 // transform trajectory to output frame
331 if (!trajectory2outputFrame(*trajectory)) {
332 return;
333 }
334 trajectory_planning_msgs::trajectory_access::setStandstill(*trajectory, true);
335 trajectory_pub_->publish(std::move(trajectory));
336 resetSolver();
337 return;
338 }
339
340 // set initial state
341 std::vector<double> x_init(*nlp_dims_->nx, 0.0);
342 if (!trajectory_planning_msgs::trajectory_access::getStandstill(latest_valid_trajectory_)) {
343 x_init = high_level_stabilization_ ? getHighLevelX0(ego_data_) : getBiLevelX0(ego_data_);
344 } else {
345 RCLCPP_WARN(this->get_logger(),
346 "Latest available trajectory is standstill. Using ego data for initial state (high-level initialization).");
347 x_init = getHighLevelX0(ego_data_);
348 }
349
350 // debug print of initial state
351 std::stringstream ss;
352 ss << "Initial state: ";
353 for (size_t i = 0; i < x_init.size(); ++i) ss << "x[" << i << "]: " << x_init[i] << (i != x_init.size() - 1 ? ", " : "");
354 RCLCPP_DEBUG(this->get_logger(), "%s", ss.str().c_str());
355
356 ocp_nlp_constraints_model_set(nlp_config_, nlp_dims_, nlp_in_, nlp_out_, 0, "lbx", x_init.data());
357 ocp_nlp_constraints_model_set(nlp_config_, nlp_dims_, nlp_in_, nlp_out_, 0, "ubx", x_init.data());
358
359 // update inputs to the ocp; skip planning cycle if update fails
360 if (!updateOcpInputs(ego_data_, object_list_, route_, reference_trajectory_, x_init)) {
361 RCLCPP_WARN(this->get_logger(), "Failed to update inputs. Skipping planning cycle.");
362 return;
363 }
364
365 // solve the optimization problem
366 int status = trajectory_optimization::acados_solve(ocp_capsule_);
367
368 // get solution
369 for (int ii = 0; ii <= nlp_dims_->N; ++ii) {
370 ocp_nlp_out_get(nlp_config_, nlp_dims_, nlp_out_, ii, "x", &xtraj_[ii * *nlp_dims_->nx]);
371 }
372 for (int ii = 0; ii < nlp_dims_->N; ++ii) {
373 ocp_nlp_out_get(nlp_config_, nlp_dims_, nlp_out_, ii, "u", &utraj_[ii * *nlp_dims_->nu]);
374 }
375
376 printSolution(status);
377 if (debug_viz_) {
378 vizCircles(viz_circles_);
379 vizEgoCircles(xtraj_, model_name_);
380 }
381
382 if (status == 1 || status == 3 || status == 4) {
383 RCLCPP_ERROR(this->get_logger(), "Solver failed with status %d.", status);
384 resetSolver();
385 return;
386 }
387
388 // convert output into trajectory message
389 convertToTrajectoryMsg(*trajectory);
390
391 bool standstill = true;
392 for (int i = 0; i < trajectory_planning_msgs::trajectory_access::getSamplePointSize(*trajectory); ++i) {
393 if (trajectory_planning_msgs::trajectory_access::getV(*trajectory, i) > standstill_threshold_) standstill = false;
394 }
395 trajectory_planning_msgs::trajectory_access::setStandstill(*trajectory, standstill);
396 if (standstill) {
397 resetSolver();
398 }
399
400 // transform trajectory to output frame
401 if (!trajectory2outputFrame(*trajectory)) {
402 return;
403 }
404
405 latest_valid_trajectory_ = *trajectory;
406 trajectory_pub_->publish(std::move(trajectory));
407 RCLCPP_INFO(this->get_logger(), "Published trajectory");
408}

410bool TrajectoryOptimizationNode::updateOcpInputs(const perception_msgs::msg::EgoData& ego_data,
411 const perception_msgs::msg::ObjectList& object_list,
412 const route_planning_msgs::msg::Route& route,
413 const trajectory_planning_msgs::msg::Trajectory& reference_trajectory,
414 const std::vector<double>& x_init) {
415 // transform inputs to target base_link frame
416 trajectory_planning_msgs::msg::Trajectory tf_reference_trajectory;
417 perception_msgs::msg::ObjectList tf_object_list;
418 route_planning_msgs::msg::Route tf_route;
419 try {
420 // reference trajectory
421 tf_reference_trajectory =
422 tf2_buffer_->transform(reference_trajectory, vehicle_frame_id_, tf2_ros::fromMsg(ego_data.header.stamp),
423 fixed_over_time_frame_id_, tf2::durationFromSec(0.01));
424 if (add_x_init_to_ref_ && trajectory_planning_msgs::trajectory_access::getX(tf_reference_trajectory, 0) > 0.0) {
425 RCLCPP_INFO(this->get_logger(), "Adding x_init to beginning of reference trajectory");
426 std::vector<double> x_0_ref = {0.0, x_init[0], x_init[1], x_init[3]};
427 tf_reference_trajectory.states.insert(tf_reference_trajectory.states.begin(), x_0_ref.begin(), x_0_ref.end());
428 }
429 // object list
430 if (!object_list.objects.empty() && object_list.header.frame_id != vehicle_frame_id_) {
431 tf_object_list = tf2_buffer_->transform(object_list, vehicle_frame_id_, tf2_ros::fromMsg(ego_data.header.stamp),
432 fixed_over_time_frame_id_, tf2::durationFromSec(0.01));
433 } else {
434 tf_object_list = object_list;
435 }
436 keepNClosestObjects(tf_object_list, static_cast<int>(p_obstacle_circles_shape_[0]));
437 // route
438 if (!route.route_elements.empty() && route.header.frame_id != vehicle_frame_id_) {
439 tf_route = tf2_buffer_->transform(route, vehicle_frame_id_, tf2_ros::fromMsg(ego_data.header.stamp),
440 fixed_over_time_frame_id_, tf2::durationFromSec(0.1));
441 } else {
442 tf_route = route;
443 }
444 } catch (tf2::TransformException& ex) {
445 RCLCPP_WARN(this->get_logger(), "Transformation is not available. Ex: %s", ex.what());
446 return false;
447 }
448
449 if (init_as_ref_ && trajectory_planning_msgs::trajectory_access::getStandstill(latest_valid_trajectory_)) {
450 // set initial guess
451 std::vector<double> initial_guess(*nlp_dims_->nx, 0.0);
452 for (int i = 0; i <= n_shots_; ++i) {
453 int idx = std::min(i, trajectory_planning_msgs::trajectory_access::getSamplePointSize(tf_reference_trajectory) - 1);
454 initial_guess[0] = trajectory_planning_msgs::trajectory_access::getX(tf_reference_trajectory, idx);
455 initial_guess[1] = trajectory_planning_msgs::trajectory_access::getY(tf_reference_trajectory, idx);
456 initial_guess[3] = trajectory_planning_msgs::trajectory_access::getV(tf_reference_trajectory, idx);
457 initial_guess[5] = trajectory_planning_msgs::trajectory_access::getTheta(tf_reference_trajectory, idx);
458 ocp_nlp_out_set(nlp_config_, nlp_dims_, nlp_out_, nlp_in_, i, "x", initial_guess.data());
459 }
460 }
461
462 // update ocp parameters
463 try {
464 this->setOcpGlobalParameters(cost_weights_, tf_reference_trajectory, tf_route);

474void TrajectoryOptimizationNode::setOcpGlobalParameters(const std::vector<double>& cost_weights,
475 const trajectory_planning_msgs::msg::Trajectory& reference_trajectory,
476 const route_planning_msgs::msg::Route& route) {
477 const auto start_time = std::chrono::steady_clock::now();
478 std::vector<double> global_params;
479
480 // cost weights
481 global_params.insert(global_params.end(), cost_weights.begin(), cost_weights.end());
482
483 // other cost params
484 global_params.push_back(thw_);
485 global_params.push_back(d_min_obstacle_long_);
486 global_params.push_back(d_min_obstacle_lat_);
487 global_params.push_back(d_min_boundary_lat_);
488
489 // reference path (including boundaries)
490 const size_t n_ref_states = static_cast<size_t>(p_ref_path_shape_[0] * p_ref_path_shape_[1]);
491 std::vector<std::pair<double, double>> boundary_distances = normalBoundaryDistance(reference_trajectory, route);
492 // fill ref vector for ocp -> psi, x, y, v, d_bound_left, d_bound_right
493 std::vector<double> ref;
494 for (int i = 0; i < trajectory_planning_msgs::trajectory_access::getSamplePointSize(reference_trajectory); ++i) {
495 ref.push_back(trajectory_planning_msgs::trajectory_access::getTheta(reference_trajectory, i));
496 ref.push_back(trajectory_planning_msgs::trajectory_access::getX(reference_trajectory, i));
497 ref.push_back(trajectory_planning_msgs::trajectory_access::getY(reference_trajectory, i));
498 ref.push_back(trajectory_planning_msgs::trajectory_access::getV(reference_trajectory, i));
499 ref.push_back(boundary_distances[i].first); // left boundary distance
500 ref.push_back(boundary_distances[i].second); // right boundary distance
501 }
502 if (ref.size() >= n_ref_states) {
503 global_params.insert(global_params.end(), ref.begin(),
504 ref.begin() + static_cast<std::vector<double>::difference_type>(n_ref_states));
505 } else {
506 // TODO: what to do here? Currently just copy the whole reference trajectory and rest is filled with infinity // NOLINT(google-readability-todo)
507 global_params.insert(global_params.end(), ref.begin(), ref.end());
508 global_params.insert(global_params.end(), n_ref_states - ref.size(), std::numeric_limits<double>::infinity());
509 }
510
511 if (global_params.size() != static_cast<size_t>(nlp_dims_->np_global)) {
512 RCLCPP_ERROR(this->get_logger(), "Size of global parameters (%zu) does not match expected size (%d).", global_params.size(),
513 nlp_dims_->np_global);
514 throw std::runtime_error("Size of global parameters does not match expected size.");
515 }
516 trajectory_optimization::acados_set_p_global_and_precompute_dependencies(ocp_capsule_, global_params.data(),
517 static_cast<int>(global_params.size()));
518 const auto elapsed_ms = std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start_time).count();
519 RCLCPP_DEBUG(this->get_logger(), "setOcpGlobalParameters duration: %.3f ms", elapsed_ms);
520}

522void TrajectoryOptimizationNode::setOcpParameters(const perception_msgs::msg::EgoData& ego_data,
523 const perception_msgs::msg::ObjectList& object_list) {
524 const auto start_time = std::chrono::steady_clock::now();
525 // loop over shooting intervals
526 double floating_dynamic_weight = 1.0;
527 double dt = optimization_horizon_ / n_shots_;
528 for (int i = 0; i <= n_shots_; ++i) {
529 // dynamic weight
530 int idx = 0;
531 int n = 1;
532 std::vector<int> idx_dynamic_weight(n);
533 // fill vector with values from idx to idx + n
534 std::iota(idx_dynamic_weight.begin(), idx_dynamic_weight.end(), idx);
535 trajectory_optimization::acados_update_params_sparse(ocp_capsule_, i, idx_dynamic_weight.data(), &floating_dynamic_weight, n);
536 floating_dynamic_weight *= dynamic_weight_;
537
538 // obstacles
539 idx += n;
540 n = static_cast<int>(p_obstacle_circles_shape_[0] * p_obstacle_circles_shape_[1]);
541 std::vector<double> circles; // [x1, y1, r1, x2, y2, r2, ...]
542
543 for (size_t j = 0; j < object_list.objects.size(); ++j) {
544 double x_tgt = 0.0, y_tgt = 0.0, yaw_tgt = 0.0;
545 std::vector<double> TIME, X, Y, YAW;
546 // TODO: should not be done for each shooting interval. Could be improved. // NOLINT(google-readability-todo)
547 TIME.push_back(static_cast<double>(rclcpp::Time(object_list.header.stamp).nanoseconds()) / 1e9);
548 X.push_back(perception_msgs::object_access::getX(object_list.objects[j]));
549 Y.push_back(perception_msgs::object_access::getY(object_list.objects[j]));
550 YAW.push_back(perception_msgs::object_access::getYaw(object_list.objects[j]));
551 if (consider_objects_ == CONSIDER_OBJECTS::PREDICTED_OBJECTS && object_list.objects[j].state_predictions.size() > 0) {
552 for (auto& predicted_state : object_list.objects[j].state_predictions[0].states) {
553 TIME.push_back(static_cast<double>(rclcpp::Time(predicted_state.header.stamp).nanoseconds()) / 1e9);
554 X.push_back(perception_msgs::object_access::getX(predicted_state));
555 Y.push_back(perception_msgs::object_access::getY(predicted_state));
556 YAW.push_back(perception_msgs::object_access::getYaw(predicted_state));
557 }
558 double des_time = static_cast<double>(rclcpp::Time(ego_data.header.stamp).nanoseconds()) / 1e9 + dt * i;
559 linearInterpolation(TIME, X, des_time, x_tgt);
560 linearInterpolation(TIME, Y, des_time, y_tgt);
561 linearInterpolation(TIME, YAW, des_time, yaw_tgt, true);
562 } else {
563 x_tgt = X.front();
564 y_tgt = Y.front();
565 yaw_tgt = YAW.front();
566 }
567 // ensure that x_tgt and y_tgt represent the geometric center of the object
568 double alpha = std::atan2(object_list.objects[j].state.reference_point.translation_to_geometric_center.y,
569 object_list.objects[j].state.reference_point.translation_to_geometric_center.x);
570 double beta = wrap_angle_rad(yaw_tgt - alpha);
571 double a = std::sqrt(std::pow(object_list.objects[j].state.reference_point.translation_to_geometric_center.x, 2) +
572 std::pow(object_list.objects[j].state.reference_point.translation_to_geometric_center.y, 2));
573 x_tgt += a * std::cos(beta);
574 y_tgt += a * std::sin(beta);
575
576 std::vector<double> obj_circles =
577 discretizeBB2Circles(x_tgt, y_tgt, yaw_tgt, perception_msgs::object_access::getLength(object_list.objects[j]),
578 perception_msgs::object_access::getWidth(object_list.objects[j]));
579
580 circles.insert(circles.end(), obj_circles.begin(), obj_circles.end());
581 if (circles.size() >= static_cast<size_t>(n)) {
582 circles.resize(n);
583 break;
584 }
585 }
586 // fill up with dummy "ghost" obstacle circles at (10000, 10000) to avoid NaNs in the optimization problem
587 // TODO: improve this // NOLINT(google-readability-todo)
588 while (circles.size() < static_cast<size_t>(n)) {
589 std::vector<double> dummy_circle = {10000.0, 10000.0, 1.0};
590 circles.insert(circles.end(), dummy_circle.begin(), dummy_circle.end());
591 }
592 if ((circles.size() % p_obstacle_circles_shape_[1]) != 0) {
593 RCLCPP_WARN(this->get_logger(), "Circles vector size is not a multiple of the circle shape. Resizing.");
594 circles.resize(n);
595 }
596 if (debug_viz_) viz_circles_.insert(viz_circles_.end(), circles.begin(), circles.end());
597
598 std::vector<int> idx_obstacles(n);
599 // fill vector with values from idx to idx + n
600 std::iota(idx_obstacles.begin(), idx_obstacles.end(), idx);
601 trajectory_optimization::acados_update_params_sparse(ocp_capsule_, i, idx_obstacles.data(), circles.data(), n);
602 }
603 const auto elapsed_ms = std::chrono::duration<double, std::milli>(std::chrono::steady_clock::now() - start_time).count();
604 RCLCPP_DEBUG(this->get_logger(), "setOcpParameters duration: %.3f ms", elapsed_ms);
605}
